# 插件扩展

## 概述

若依前端提供了 5 个全局插件，封装了常用的业务功能，通过 `app.config.globalProperties` 挂载到全局，可在组件中通过 `this.$xxx` 访问。

**插件列表**:

```
src/plugins/
├── index.js          # 插件注册入口
├── auth.js           # 认证权限插件
├── cache.js          # 缓存插件
├── download.js       # 文件下载插件
├── modal.js          # 消息提示插件
└── tab.js            # 标签页操作插件
```

---

## 全局注册

**源码**: [`plugins/index.js`](../../ruoyi-ui/src/plugins/index.js)

```javascript
import tab from './tab'
import auth from './auth'
import cache from './cache'
import modal from './modal'
import download from './download'

export default function installPlugins(app){
  app.config.globalProperties.$tab = tab      // 标签页操作
  app.config.globalProperties.$auth = auth    // 认证权限
  app.config.globalProperties.$cache = cache  // 缓存
  app.config.globalProperties.$modal = modal  // 消息提示
  app.config.globalProperties.$download = download  // 文件下载
}
```

在 `main.js` 中调用：

```javascript
import installPlugins from './plugins'

const app = createApp(App)
installPlugins(app)
```

## $modal - 消息提示插件

**源码**: [`plugins/modal.js`](../../ruoyi-ui/src/plugins/modal.js)

封装 Element Plus 的消息提示组件，提供统一的调用方式。

### API 列表

| 方法 | 参数 | 返回值 | 说明 |
|------|------|--------|------|
| `msg(content)` | content: String | - | 普通消息提示 |
| `msgError(content)` | content: String | - | 错误消息 |
| `msgSuccess(content)` | content: String | - | 成功消息 |
| `msgWarning(content)` | content: String | - | 警告消息 |
| `alert(content)` | content: String | Promise | Alert 对话框 |
| `alertError(content)` | content: String | Promise | 错误 Alert |
| `alertSuccess(content)` | content: String | Promise | 成功 Alert |
| `alertWarning(content)` | content: String | Promise | 警告 Alert |
| `notify(content)` | content: String | - | 通知提示 |
| `notifyError(content)` | content: String | - | 错误通知 |
| `notifySuccess(content)` | content: String | - | 成功通知 |
| `notifyWarning(content)` | content: String | - | 警告通知 |
| `confirm(content)` | content: String | Promise | 确认对话框 |
| `prompt(content)` | content: String | Promise | 输入对话框 |
| `loading(content)` | content: String | - | 打开加载遮罩 |
| `closeLoading()` | - | - | 关闭加载遮罩 |

### 使用示例

#### 消息提示

```javascript
// 在组件中
export default {
  methods: {
    handleSubmit() {
      // 成功提示
      this.$modal.msgSuccess('保存成功')
      
      // 错误提示
      this.$modal.msgError('保存失败，请稍后重试')
      
      // 警告提示
      this.$modal.msgWarning('该操作会影响其他模块')
      
      // 普通提示
      this.$modal.msg('操作已完成')
    }
  }
}
```

#### Composition API 中使用

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()

const handleSubmit = () => {
  proxy.$modal.msgSuccess('保存成功')
}
</script>
```

#### Alert 对话框

```javascript
// 基础用法
this.$modal.alert('操作成功')

// 带类型
this.$modal.alertSuccess('操作成功')
this.$modal.alertError('操作失败')
this.$modal.alertWarning('该操作不可恢复')

// 等待用户关闭
await this.$modal.alert('请阅读后确认')
console.log('用户已关闭弹窗')
```

#### Confirm 确认框

```javascript
// 删除操作
this.$modal.confirm('确定要删除选中的数据吗？')
  .then(() => {
    return deleteUser(userId)
  })
  .then(() => {
    this.$modal.msgSuccess('删除成功')
  })
  .catch(() => {
    // 用户取消
  })
```

#### Prompt 输入框

```javascript
// 输入对话框
this.$modal.prompt('请输入备注信息：')
  .then(({ value }) => {
    console.log('用户输入：', value)
    return submitRemark(value)
  })
  .then(() => {
    this.$modal.msgSuccess('提交成功')
  })
  .catch(() => {
    // 用户取消
  })
```

#### Loading 加载遮罩

```javascript
// 打开加载遮罩
this.$modal.loading('正在处理，请稍候...')

// 执行异步操作
someAsyncOperation()
  .then(() => {
    this.$modal.msgSuccess('处理成功')
  })
  .catch((err) => {
    this.$modal.msgError(err.message)
  })
  .finally(() => {
    // 关闭加载遮罩
    this.$modal.closeLoading()
  })
```

### 最佳实践

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()

// 统一的错误处理
const handleError = (error) => {
  proxy.$modal.msgError(error.message || '操作失败')
  proxy.$modal.closeLoading()
}

// 统一的成功处理
const handleSuccess = (message = '操作成功') => {
  proxy.$modal.msgSuccess(message)
  proxy.$modal.closeLoading()
}

// 使用示例
const handleSubmit = async () => {
  proxy.$modal.loading('正在提交...')
  try {
    await submitForm()
    handleSuccess()
  } catch (error) {
    handleError(error)
  }
}
</script>
```

## $cache - 缓存插件

**源码**: [`plugins/cache.js`](../../ruoyi-ui/src/plugins/cache.js)

封装 sessionStorage 和 localStorage 的缓存操作。

### API 列表

#### session - 会话级缓存

| 方法 | 参数 | 返回值 | 说明 |
|------|------|--------|------|
| `set(key, value)` | key: String, value: String | - | 设置缓存 |
| `get(key)` | key: String | String/null | 获取缓存 |
| `setJSON(key, jsonValue)` | key: String, jsonValue: Object | - | 设置 JSON 缓存 |
| `getJSON(key)` | key: String | Object/null | 获取 JSON 缓存 |
| `remove(key)` | key: String | - | 删除缓存 |

#### local - 本地缓存

同上，区别在于使用 localStorage 持久化存储。

### 使用示例

#### 基础用法

```javascript
// SessionStorage
this.$cache.session.set('username', 'admin')
const username = this.$cache.session.get('username')
this.$cache.session.remove('username')

// Localstorage
this.$cache.local.set('theme', 'dark')
const theme = this.$cache.local.get('theme')
this.$cache.local.remove('theme')
```

#### JSON 数据存取

```javascript
// 存储对象
const user = { id: 1, name: 'admin', roles: ['admin'] }
this.$cache.session.setJSON('currentUser', user)

// 获取对象
const currentUser = this.$cache.session.getJSON('currentUser')
console.log(currentUser.name)  // 'admin'

// 存储数组
const permissions = ['system:user:list', 'system:user:add']
this.$cache.local.setJSON('permissions', permissions)

// 获取数组
const perms = this.$cache.local.getJSON('permissions')
```

#### Composition API 中使用

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()

// 保存查询条件
const saveQueryParams = (params) => {
  proxy.$cache.local.setJSON('userQueryParams', params)
}

// 恢复查询条件
const restoreQueryParams = () => {
  return proxy.$cache.local.getJSON('userQueryParams') || {}
}

// 切换标签页时保存
const handleTabChange = (tabName) => {
  proxy.$cache.session.set('activeTab', tabName)
}
</script>
```

#### 实际应用场景

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()

// 记住上次的分页参数
const queryParams = ref(proxy.$cache.local.getJSON('userQueryParams') || {
  pageNum: 1,
  pageSize: 10
})

// 查询时保存参数
const handleQuery = () => {
  proxy.$cache.local.setJSON('userQueryParams', queryParams.value)
  getList(queryParams.value)
}

// 重置时清除缓存
const handleReset = () => {
  proxy.$cache.local.remove('userQueryParams')
  queryParams.value = { pageNum: 1, pageSize: 10 }
  handleQuery()
}
</script>
```

## $auth - 认证权限插件

**源码**: [`plugins/auth.js`](../../ruoyi-ui/src/plugins/auth.js)

提供权限和角色的验证方法。

### API 列表

| 方法 | 参数 | 返回值 | 说明 |
|------|------|--------|------|
| `hasPermi(permission)` | permission: String | Boolean | 验证是否有某权限 |
| `hasPermiOr(permissions)` | permissions: String[] | Boolean | 验证是否有任一权限 |
| `hasPermiAnd(permissions)` | permissions: String[] | Boolean | 验证是否有全部权限 |
| `hasRole(role)` | role: String | Boolean | 验证是否有某角色 |
| `hasRoleOr(roles)` | roles: String[] | Boolean | 验证是否有任一角色 |
| `hasRoleAnd(roles)` | roles: String[] | Boolean | 验证是否有全部角色 |

### 使用示例

#### 单个权限判断

```javascript
// 在组件中
if (this.$auth.hasPermi('system:user:add')) {
  // 有新增用户权限
}

if (this.$auth.hasRole('admin')) {
  // 有 admin 角色
}
```

#### 多个权限判断

```javascript
// 任一权限
const canEdit = this.$auth.hasPermiOr([
  'system:user:add',
  'system:user:edit'
])

// 全部权限
const canManage = this.$auth.hasPermiAnd([
  'system:user:add',
  'system:user:edit',
  'system:user:remove'
])

// 任一角色
const isAdminOrManager = this.$auth.hasRoleOr(['admin', 'manager'])

// 全部角色
const isAdminAndUser = this.$auth.hasRoleAnd(['admin', 'user'])
```

#### Composition API 中使用

```vue
<script setup>
import { getCurrentInstance, computed } from 'vue'

const { proxy } = getCurrentInstance()

// 权限判断
const canAddUser = computed(() => 
  proxy.$auth.hasPermi('system:user:add')
)

const canExportData = computed(() =>
  proxy.$auth.hasPermiOr(['system:user:export', 'system:dept:export'])
)

const isSuperAdmin = computed(() =>
  proxy.$auth.hasRole('admin')
)
</script>

<template>
  <el-button v-if="canAddUser" @click="handleAdd">新增</el-button>
  <el-button v-if="canExportData" @click="handleExport">导出</el-button>
</template>
```

## $tab - 标签页操作插件

**源码**: [`plugins/tab.js`](../../ruoyi-ui/src/plugins/tab.js)

提供标签页的刷新、关闭、切换等操作。

### API 列表

| 方法 | 参数 | 返回值 | 说明 |
|------|------|--------|------|
| `refreshPage(obj)` | obj: Object (可选) | Promise | 刷新当前页 |
| `closeOpenPage(obj)` | obj: Object (可选) | - | 关闭当前页并打开新页 |
| `closePage(obj)` | obj: Object (可选) | Promise | 关闭指定页签 |
| `closeAllPage()` | - | Promise | 关闭所有页签 |
| `closeLeftPage(obj)` | obj: Object (可选) | Promise | 关闭左侧页签 |
| `closeRightPage(obj)` | obj: Object (可选) | Promise | 关闭右侧页签 |
| `closeOtherPage(obj)` | obj: Object (可选) | Promise | 关闭其他页签 |
| `openPage(title, url, params)` | title: String, url: String, params: Object | - | 打开页签 |
| `updatePage(obj)` | obj: Object | Promise | 修改页签信息 |

### 使用示例

#### 刷新页面

```javascript
// 刷新当前页
this.$tab.refreshPage()

// 刷新指定页面
this.$tab.refreshPage({ 
  path: '/system/user',
  query: { id: 1 }
})

// Composition API
const { proxy } = getCurrentInstance()
await proxy.$tab.refreshPage()
```

#### 关闭页签

```javascript
// 关闭当前页
this.$tab.closePage()

// 关闭指定页签
this.$tab.closePage({ path: '/system/user' })

// 关闭所有页签
this.$tab.closeAllPage()

// 关闭左侧页签
this.$tab.closeLeftPage()

// 关闭右侧页签
this.$tab.closeRightPage()

// 关闭其他页签
this.$tab.closeOtherPage()
```

#### 打开新页签

```javascript
// 打开新页签
this.$tab.openPage('用户详情', '/system/user/detail', { id: 1 })

// 关闭当前页并打开新页
this.$tab.closeOpenPage({ path: '/system/user' })
```

#### 实际应用场景

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()
const router = useRouter()

// 新增成功后刷新列表
const handleAdd = async () => {
  await submitForm()
  proxy.$modal.msgSuccess('新增成功')
  await proxy.$tab.refreshPage()
}

// 删除成功后关闭详情页返回列表
const handleDelete = async () => {
  await proxy.$modal.confirm('确定删除吗？')
  await deleteUser(userId)
  proxy.$modal.msgSuccess('删除成功')
  await proxy.$tab.closePage()
  router.push('/system/user')
}

// 编辑成功后关闭弹窗刷新列表
const handleEdit = async () => {
  await submitForm()
  proxy.$modal.msgSuccess('修改成功')
  await proxy.$tab.refreshPage()
}
</script>
```

## $download - 文件下载插件

**源码**: [`plugins/download.js`](../../ruoyi-ui/src/plugins/download.js)

提供文件下载功能，支持带 Token 认证下载。

### API 列表

| 方法 | 参数 | 说明 |
|------|------|------|
| `name(fileName, isDelete)` | fileName: String, isDelete: Boolean | 下载服务器文件 |
| `resource(resource)` | resource: String | 下载资源文件 |
| `zip(url, name)` | url: String, name: String | 下载 ZIP 文件 |
| `saveAs(text, name, opts)` | text: Blob, name: String, opts: Object | 保存文件 |

### 使用示例

#### 下载服务器文件

```javascript
// 下载并删除源文件
this.$download.name('export.xlsx', true)

// 下载不删除
this.$download.name('report.pdf', false)
```

#### 下载资源

```javascript
// 下载头像
this.$download.resource('/profile/avatar.jpg')

// 下载附件
this.$download.resource('/attach/2026/04/file.pdf')
```

#### 下载 ZIP

```javascript
// 下载导入模板
this.$download.zip('/system/user/importTemplate', '用户模板.zip')

// 下载导出文件
this.$download.zip('/system/user/export?deptId=100', '部门用户数据.zip')
```

#### Composition API 中使用

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()

// 导出用户数据
const handleExport = () => {
  proxy.$download.zip(
    '/system/user/export?' + new URLSearchParams(queryParams.value).toString(),
    '用户数据.xlsx'

const downloadTemplate = () => {
  proxy.$download.zip('/system/user/importTemplate', '导入模板.xlsx')
}
</script>

<template>
  <el-button @click="handleExport">导出</el-button>
  <el-button @click="downloadTemplate">下载模板</el-button>
</template>
```

#### 配合 API 下载

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()

  const url = '/system/user/export?' + new URLSearchParams(queryParams.value).toString()
  proxy.$download.zip(url, '用户数据.xlsx')
```

## 插件组合使用

### 完整的 CRUD 操作

```vue
<script setup>
import { getCurrentInstance } from 'vue'

const { proxy } = getCurrentInstance()

// 新增
const handleAdd = async () => {
  proxy.$modal.loading('正在提交...')

    await addUser(form.value)
    proxy.$modal.msgSuccess('新增成功')
    await proxy.$tab.refreshPage()
  } catch (error) {
    proxy.$modal.msgError(error.message)

    '用户数据.xlsx'
  )
}
</script>
```

```vue
import { getCurrentInstance, computed } from 'vue'
```

## 最佳实践

### 1. 统一错误处理

```javascript
// utils/error-handler.js
export function handleError(error, modal) {
  console.error(error)
  modal.msgError(error.message || '操作失败，请稍后重试')
  modal.closeLoading()
}

// 使用
const handleSubmit = async () => {
  modal.loading('正在提交...')
  try {
    await api.submit()
    modal.msgSuccess('提交成功')
  } catch (error) {
    handleError(error, modal)
  }
}
```

### 2. 统一成功处理

```javascript
// utils/success-handler.js
export function handleSuccess(modal, tab, message = '操作成功', refresh = true) {
  modal.msgSuccess(message)
  if (refresh) {
    return tab.refreshPage()
  }
  return Promise.resolve()
}
```

### 3. 权限缓存

```javascript
// 缓存权限数据避免重复请求
const permissions = proxy.$cache.session.getJSON('userPermissions')
if (!permissions) {
  await userStore.getInfo()
  proxy.$cache.session.setJSON('userPermissions', userStore.permissions)
}
```

### 4. 下载统一处理

```javascript
// utils/download-helper.js
export function downloadFile(url, filename, params = {}) {
  const query = new URLSearchParams(params).toString()
  const fullUrl = `${url}?${query}`
  return download.zip(fullUrl, filename)
}

// 使用
const handleExport = () => {
  downloadFile('/system/user/export', '用户数据.xlsx', queryParams.value)
}
```
